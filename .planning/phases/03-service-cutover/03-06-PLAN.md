    ```ts
    import { NextResponse } from "next/server";
    import { sql } from "drizzle-orm";
    import { z } from "zod";
    import { db } from "@/lib/db/client";
    import { wisenetFetch } from "@/lib/wisenet/client";
    import { getAuthEnv, getWisenetEnv } from "@/lib/runtime/env";

    export const runtime = "nodejs";
    // SVC-07 / D-32: NO 'use cache: remote' — health must probe live every call.
    // SVC-07 / D-32: NO requireSessionUser — uptime monitors must reach this endpoint without auth.

    interface SubsystemStatus {
      status: "ok" | "degraded" | "down";
      latencyMs?: number;
      error?: string;
    }

    interface AuthStatus {
      status: "ok" | "down";
      error?: string;
    }

    export async function GET() {
      const timestamp = new Date().toISOString();
      const [wisenet, postgres, auth] = await Promise.all([
        probeWisenet(),
        probePostgres(),
        probeAuth(),
      ]);

      // D-32 rollup rule:
      //   any "down" → "down"
      //   any "degraded" (no downs) → "degraded"
      //   else → "ok"
      const overall: "ok" | "degraded" | "down" =
        wisenet.status === "down" || postgres.status === "down" || auth.status === "down"
          ? "down"
          : wisenet.status === "degraded" || postgres.status === "degraded"
            ? "degraded"
            : "ok";

      const statusCode = overall === "down" ? 503 : 200; // D-32 line 161

      return NextResponse.json(
        {
          status: overall,
          timestamp,
          subsystems: { wisenet, postgres, auth },
          deployedAt: process.env.VERCEL_DEPLOYMENT_ID,
          // deploymentId deliberately omitted — Phase 4 DEPL-06 formalizes via X-BG-Deploy-Id header.
        },
        { status: statusCode },
      );
    }

    async function probeWisenet(): Promise<SubsystemStatus> {
      const start = Date.now();
      try {
        const env = getWisenetEnv();
        await wisenetFetch(
          `/institutes/v3/${env.WISENET_CENTER_ID}/students?page_number=1&page_size=1`,
          z.object({ status: z.union([z.string(), z.number()]) }).passthrough(),
          { signal: AbortSignal.timeout(5_000) }, // tighter than client default 15s — health must fail fast
        );
        return { status: "ok", latencyMs: Date.now() - start };
      } catch (error) {
        return {
          status: "down",
          latencyMs: Date.now() - start,
          // WisenetError.message is "Wisenet ${status} at ${path}" — safe to surface (no creds).
          error: error instanceof Error ? error.message : String(error),
        };
      }
    }

    async function probePostgres(): Promise<SubsystemStatus> {
      const start = Date.now();
      try {
        await db.execute(sql`SELECT 1`); // D-32: HTTP driver, not Pool
        return { status: "ok", latencyMs: Date.now() - start };
      } catch (error) {
        return {
          status: "down",
          latencyMs: Date.now() - start,
          error: error instanceof Error ? error.message : String(error),
        };
      }
    }

    function probeAuth(): AuthStatus {
      try {
        getAuthEnv(); // throws if any required env var is missing
        return { status: "ok" };
      } catch (error) {
        return {
          status: "down",
          error: error instanceof Error ? error.message : String(error),
        };
      }
    }
    ```

    ```ts
    import { describe, expect, it, vi, beforeEach } from "vitest";

    const wisenetFetch = vi.fn();
    const dbExecute = vi.fn();
    const getAuthEnv = vi.fn();

    vi.mock("@/lib/wisenet/client", () => ({ wisenetFetch }));
    vi.mock("@/lib/db/client", () => ({ db: { execute: dbExecute } }));
    vi.mock("@/lib/runtime/env", () => ({
      getAuthEnv,
      getWisenetEnv: () => ({
        WISENET_BASE_URL: "https://api.wiseapp.live",
        WISENET_USER_ID: "test",
        WISENET_API_KEY: "test",
        WISENET_CENTER_ID: "test-center",
        WISENET_NAMESPACE: "begifted-education",
      }),
    }));

    describe("/api/health route", () => {
      beforeEach(() => {
        vi.clearAllMocks();
      });

      it("returns 200 with status=ok when all subsystems are healthy", async () => {
        wisenetFetch.mockResolvedValue({ status: "OK" });
        dbExecute.mockResolvedValue({});
        getAuthEnv.mockReturnValue({});

        const { GET } = await import("@/app/api/health/route");
        const response = await GET();
        const body = await response.json();

        expect(response.status).toBe(200);
        expect(body.status).toBe("ok");
        expect(body.subsystems.wisenet.status).toBe("ok");
        expect(body.subsystems.postgres.status).toBe("ok");
        expect(body.subsystems.auth.status).toBe("ok");
        expect(body.timestamp).toMatch(/^\d{4}-\d{2}-\d{2}T/);
      });

      it("returns 503 with status=down when Postgres is unreachable", async () => {
        wisenetFetch.mockResolvedValue({ status: "OK" });
        dbExecute.mockRejectedValue(new Error("connection terminated"));
        getAuthEnv.mockReturnValue({});

        const { GET } = await import("@/app/api/health/route");
        const response = await GET();
        const body = await response.json();

        expect(response.status).toBe(503);
        expect(body.status).toBe("down");
        expect(body.subsystems.postgres.status).toBe("down");
        expect(body.subsystems.postgres.error).toContain("connection terminated");
        expect(body.subsystems.wisenet.status).toBe("ok");
      });

      it("returns 503 with status=down when Wisenet probe fails", async () => {
        wisenetFetch.mockRejectedValue(new Error("Wisenet 500 at /institutes/v3/test-center/students"));
        dbExecute.mockResolvedValue({});
        getAuthEnv.mockReturnValue({});

        const { GET } = await import("@/app/api/health/route");
        const response = await GET();
        const body = await response.json();

        expect(response.status).toBe(503);
        expect(body.status).toBe("down");
        expect(body.subsystems.wisenet.status).toBe("down");
        // Security: WisenetError message format is safe (no creds, just status + path).
        expect(body.subsystems.wisenet.error).toContain("Wisenet");
        expect(body.subsystems.wisenet.error).not.toContain("api_key");
        expect(body.subsystems.wisenet.error).not.toContain("apiKey");
      });

      it("returns 503 with status=down when auth env is missing", async () => {
        wisenetFetch.mockResolvedValue({ status: "OK" });
        dbExecute.mockResolvedValue({});
        getAuthEnv.mockImplementation(() => {
          throw new Error("Missing required environment variable: NEXTAUTH_SECRET");
        });

        const { GET } = await import("@/app/api/health/route");
        const response = await GET();
        const body = await response.json();

        expect(response.status).toBe(503);
        expect(body.status).toBe("down");
        expect(body.subsystems.auth.status).toBe("down");
        expect(body.subsystems.auth.error).toContain("NEXTAUTH_SECRET");
      });

      it("populates deployedAt from VERCEL_DEPLOYMENT_ID env var when present", async () => {
        const previous = process.env.VERCEL_DEPLOYMENT_ID;
        process.env.VERCEL_DEPLOYMENT_ID = "dpl_abc123";
        wisenetFetch.mockResolvedValue({ status: "OK" });
        dbExecute.mockResolvedValue({});
        getAuthEnv.mockReturnValue({});

        try {
          const { GET } = await import("@/app/api/health/route");
          const response = await GET();
          const body = await response.json();
          expect(body.deployedAt).toBe("dpl_abc123");
        } finally {
          if (previous === undefined) delete process.env.VERCEL_DEPLOYMENT_ID;
          else process.env.VERCEL_DEPLOYMENT_ID = previous;
        }
      });
    });
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Public internet → /api/health (UNAUTHENTICATED) | Anyone on the internet can reach this endpoint. Surface area is intentionally minimal: 3 probes, structured JSON, no body parsing, no DB writes, no Wisenet writes. |
| /api/health → Wisenet probe | Network egress; passes auth headers via `wisenetFetch`. |
| /api/health → Postgres probe | HTTP driver; `SELECT 1` is read-only and parameter-free. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-23 | Information Disclosure | Wisenet credentials surfaced in `subsystems.wisenet.error` | mitigate | Phase 2 02-VERIFICATION confirmed `WisenetError.message = "Wisenet ${status} at ${path}"`. Test 3 in Task 2 pins this contract with `not.toContain("api_key")` assertion. If Phase 4 introduces a different error class, the test fails loudly. |
| T-03-24 | Denial of Service | Public unauthenticated endpoint hit at high frequency | accept | 5s Wisenet timeout caps per-probe latency; Postgres `SELECT 1` is sub-ms. Vercel rate limiting + Postgres connection-pool guard (Phase 2 max:3 singleton) handle burst load. Phase 4 DEPL-04 may add explicit rate limiting if uptime-monitor traffic becomes problematic. |
