    import { describe, it, expect, vi, beforeEach, afterEach } from "vitest";

    // Helper: create a Response with a ReadableStream body split into 3 chunks
    function makeChunkedJsonResponse(json: unknown): Response {
      const fullText = JSON.stringify(json);
      const encoder = new TextEncoder();
      const chunks = [
        encoder.encode(fullText.slice(0, Math.floor(fullText.length / 3))),
        encoder.encode(fullText.slice(Math.floor(fullText.length / 3), Math.floor((2 * fullText.length) / 3))),
        encoder.encode(fullText.slice(Math.floor((2 * fullText.length) / 3))),
      ];
      const stream = new ReadableStream({
        start(controller) {
          for (const chunk of chunks) controller.enqueue(chunk);
          controller.close();
        },
      });
      return new Response(stream, {
        status: 200,
        headers: { "Content-Type": "application/json" },
      });
    }

    describe("chunked-transfer regression (TEST-06 reframe)", () => {
      // Env snapshot for Wisenet + DB env vars
      const ORIGINAL_ENV = { ...process.env };
      // Map of required Wisenet fixture responses per endpoint
      // Each wisenetFetch call hits a different path — the mock must respond correctly to each

      beforeEach(() => {
        // Set required Wisenet env vars
        process.env.WISENET_BASE_URL = "https://api.wiseapp.live";
        process.env.WISENET_USER_ID = "test-user-id";
        process.env.WISENET_API_KEY = "test-api-key";
        process.env.WISENET_CENTER_ID = "test-center-id";
        process.env.WISENET_NAMESPACE = "test-ns";
        vi.resetModules();
      });

      afterEach(() => {
        vi.unstubAllGlobals();
        vi.restoreAllMocks();
        process.env = { ...ORIGINAL_ENV };
      });

      it("getDashboardPayload handles chunked Wisenet responses without error", async () => {
        // Load fixture data that represents a valid Wisenet students list response
        const studentsFixture = await import("./fixtures/wisenet/students_list_page1.json");

        // Stub fetch to return chunked responses for all Wisenet requests
        // Use a counter or URL matching to return appropriate fixture per endpoint
        vi.stubGlobal("fetch", vi.fn().mockImplementation((_url: string) => {
          // All endpoints return their fixture data as chunked responses
          return Promise.resolve(makeChunkedJsonResponse(studentsFixture));
        }));

        // Also mock DB calls (not testing DB here — testing chunked response handling)
        vi.mock("@/lib/db/queries", () => ({
          loadActionStateMap: vi.fn().mockResolvedValue(new Map()),
          listInactive: vi.fn().mockResolvedValue([]),
          bulkGetAdminOwnership: vi.fn().mockResolvedValue(new Map()),
        }));

        const { getDashboardPayload } = await import("@/lib/dashboard/service");
        const payload = await getDashboardPayload();

        // Assert pipeline resolved without error — payload has required shape
        expect(payload).toBeTruthy();
        expect(typeof payload.students).toBe("object");
        // Assert we got through the mapper without errors (students may be empty if fixture doesn't map)
      });
    });
