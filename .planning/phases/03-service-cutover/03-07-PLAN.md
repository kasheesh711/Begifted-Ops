    ```ts
    import { describe, expect, it, beforeEach, afterEach, vi } from "vitest";

    /**
     * TEST-06 — chunked-transfer regression (REFRAMED per D-34)
     *
     * Original intent (Phase 2): simulate Apps Script chunked transport → assert dashboard renders.
     * New intent (Phase 3): simulate HTTP chunked-encoded response from Wisenet mock → assert
     * wisenetFetch + Zod boundary parses successfully across chunk boundaries.
     *
     * Threat addressed (CLAUDE.md HTTP -1 deploy-state class): post-cutover, HTTP -1 means a
     * Vercel function deploy issue OR a Wisenet upstream timeout. This test covers the
     * Wisenet-side streaming-response case. The other half (Vercel deploy state) is
     * Phase 4 DEPL-06 (X-BG-Deploy-Id header) territory.
     */

    let envSnapshot: Record<string, string | undefined>;

    function makeChunkedJsonResponse(payload: unknown, chunkCount = 3): Response {
      const fullText = JSON.stringify(payload);
      const encoder = new TextEncoder();

      // Deliberate N-chunk split (default 3 per RESEARCH §Pattern 9 / 03-PATTERNS line 1010-1018):
      // exercises the chunk-concatenation path inside Response.json().
      const sliceLen = Math.ceil(fullText.length / chunkCount);
      const chunks: Uint8Array[] = [];
      for (let i = 0; i < chunkCount; i++) {
        const start = i * sliceLen;
        const end = Math.min(start + sliceLen, fullText.length);
        chunks.push(encoder.encode(fullText.slice(start, end)));
      }

      const stream = new ReadableStream({
        start(controller) {
          for (const c of chunks) controller.enqueue(c);
          controller.close();
        },
      });
      return new Response(stream, {
        status: 200,
        headers: { "Content-Type": "application/json" },
      });
    }

    describe("chunked-transfer regression (TEST-06)", () => {
      beforeEach(() => {
        vi.stubGlobal("fetch", vi.fn());
        envSnapshot = {
          WISENET_BASE_URL: process.env.WISENET_BASE_URL,
          WISENET_USER_ID: process.env.WISENET_USER_ID,
          WISENET_API_KEY: process.env.WISENET_API_KEY,
          WISENET_CENTER_ID: process.env.WISENET_CENTER_ID,
          WISENET_NAMESPACE: process.env.WISENET_NAMESPACE,
        };
        process.env.WISENET_BASE_URL = "https://api.wiseapp.live";
        process.env.WISENET_USER_ID = "test-user";
        process.env.WISENET_API_KEY = "test-key";
        process.env.WISENET_CENTER_ID = "test-center";
        process.env.WISENET_NAMESPACE = "begifted-education";
      });

      afterEach(() => {
        vi.unstubAllGlobals();
        vi.restoreAllMocks();
        for (const [k, v] of Object.entries(envSnapshot)) {
          if (v === undefined) delete process.env[k];
          else process.env[k] = v;
        }
      });

      it("wisenetFetch parses a 3-chunk ReadableStream response correctly", async () => {
        const expectedPayload = {
          status: 200,
          message: "OK",
          data: {
            students: [
              { _id: "abc123", firstName: "Jade", lastName: "Lim", parentIds: [] },
              { _id: "def456", firstName: "Gina", lastName: "Ho", parentIds: [] },
            ],
            count: 2,
          },
        };

        const fetchMock = vi.mocked(global.fetch);
        fetchMock.mockResolvedValue(makeChunkedJsonResponse(expectedPayload, 3));

        const { wisenetFetch } = await import("@/lib/wisenet/client");
        const { z } = await import("zod");

        const schema = z.object({
          status: z.union([z.string(), z.number()]),
          data: z.object({ count: z.number() }).passthrough(),
        }).passthrough();

        const parsed = await wisenetFetch("/institutes/v3/test-center/students", schema);
        expect(parsed.data.count).toBe(2);
      });

      it("survives 5-chunk split (more chunks than 3 per RESEARCH ReadableStream pattern)", async () => {
        const fetchMock = vi.mocked(global.fetch);
        const payload = { status: 200, data: { records: Array.from({ length: 50 }, (_, i) => ({ id: i })) } };
        fetchMock.mockResolvedValue(makeChunkedJsonResponse(payload, 5));

        const { wisenetFetch } = await import("@/lib/wisenet/client");
        const { z } = await import("zod");

        const schema = z.object({
          status: z.union([z.string(), z.number()]),
          data: z.object({ records: z.array(z.object({ id: z.number() })) }),
        });

        const parsed = await wisenetFetch("/institutes/v3/test-center/data", schema);
        expect(parsed.data.records).toHaveLength(50);
      });

      it("propagates WisenetError when chunked stream contains malformed JSON across boundary", async () => {
        const fetchMock = vi.mocked(global.fetch);
        // A chunked Response whose concatenated body is invalid JSON
        const stream = new ReadableStream({
          start(controller) {
            const enc = new TextEncoder();
            controller.enqueue(enc.encode('{"data":{"part'));
            controller.enqueue(enc.encode('ial":"json"'));
            controller.enqueue(enc.encode("not valid }}}"));
            controller.close();
          },
        });
        fetchMock.mockResolvedValue(
          new Response(stream, { status: 200, headers: { "Content-Type": "application/json" } }),
        );

        const { wisenetFetch } = await import("@/lib/wisenet/client");
        const { z } = await import("zod");

        await expect(
          wisenetFetch("/institutes/v3/test-center/students", z.object({ status: z.any() })),
        ).rejects.toThrow();
      });
    });
    ```
