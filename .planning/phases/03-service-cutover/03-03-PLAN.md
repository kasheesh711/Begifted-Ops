    ```ts
    import { cacheLife, cacheTag } from "next/cache";
    import { buildDashboardSourcesFromWisenet } from "@/lib/wisenet/mappers";
    import {
      buildActiveStudentSet,
      buildExcludedPackageReasons,
      buildStudentAdminOwnershipMap,
      buildPendingDeductionContext,
      buildUpcomingSessionMap,
      buildDashboardStudents,
      attachActionStatesToStudents,

    import { buildDashboardModel } from "@/lib/dashboard/analytics";
    import { getTodayDate } from "@/lib/dashboard/helpers";
    import { loadActionStateMap, listInactive, clearInactive, bulkGetAdminOwnership } from "@/lib/db/queries";
    import { DASHBOARD_CACHE_TAG } from "@/lib/dashboard/config";
    import { recordSheetsCheck, recordPayloadBuild } from "@/lib/dashboard/health-state";
    import { log } from "@/lib/runtime/logger";
    import type { DashboardPayload } from "@/types/dashboard";

    const PAYLOAD_SIZE_WARN_BYTES = 1_800_000; // RESEARCH §Pattern 7 / Critical Finding #4 — 2 MB Vercel cache item limit

    export async function getDashboardPayload(now: Date = new Date()): Promise<DashboardPayload> {
      "use cache: remote";
      cacheTag(DASHBOARD_CACHE_TAG);
      cacheLife({ stale: 60, revalidate: 60, expire: 300 }); // explicit triple per Research Critical Finding #2 — overrides D-28 shorthand

      const startedAt = Date.now();
      const today = getTodayDate(now);
      try {
        const sources = await buildDashboardSourcesFromWisenet(today);
        recordSheetsCheck(true, now.toISOString());

        const activeStudents = buildActiveStudentSet(sources.students);
        const excludedPackageReasons = buildExcludedPackageReasons(sources.studentsCourses);

        // D-06 admin ownership: Postgres sidecar overrides mapper's empty Admin field after compose
        const studentKeys = Array.from(activeStudents.keys()); // Set<string> per packages.ts contract — adapt iteration if Map
        const adminOwnership = await bulkGetAdminOwnership(studentKeys);
        const adminOwnershipMap = buildStudentAdminOwnershipMap(sources.remainingCredits, adminOwnership);

        const pendingDeductionContext = buildPendingDeductionContext(
          sources.aggregations, sources.creditControl, sources.upcomingSessions, today,
        );
        const upcomingSessionMap = buildUpcomingSessionMap(sources.upcomingSessions, today);

        const students = buildDashboardStudents(
          activeStudents, excludedPackageReasons, adminOwnershipMap,
          pendingDeductionContext, upcomingSessionMap, today,
        );

        const actionStates = await loadActionStateMap();
        attachActionStatesToStudents(students, today, actionStates);

        // Auto-reactivation: students that reappear with active packages get their inactive flag cleared.
        // Phase 2 lib/db/queries.ts::clearInactive replaces the prior Sheets-side delete.
        const inactiveKeys = await listInactive();
        const filteredStudents = [];
        for (const s of students) {
          if (inactiveKeys.has(s.studentKey) && s.packages.some((p) => p.status !== "nodata")) {
            await clearInactive(s.studentKey);
          } else if (inactiveKeys.has(s.studentKey)) {
            continue; // skip students still flagged inactive with no active packages
          }
          filteredStudents.push(s);
        }

        // Snapshot state passed as null per Plan 03-08 (snapshot-store deleted, deltas reset on cold start
        // — accepted tradeoff per CONTEXT §code_context line 447).
        const dashboardModel = buildDashboardModel(filteredStudents, null, today, now);
        recordPayloadBuild(Date.now() - startedAt, now.toISOString());

        // RESEARCH Critical Finding #4: 2 MB Vercel cache item limit. Warn early.
        const serialized = JSON.stringify(dashboardModel.payload);
        if (serialized.length > PAYLOAD_SIZE_WARN_BYTES) {
          log("warn", "service.ts/getDashboardPayload",
              new Error("payload-size-near-2mb-cache-limit"),
              { sizeBytes: serialized.length });
        }

        return dashboardModel.payload;
      } catch (error) {
        log("error", "service.ts/getDashboardPayload", error);
        throw error;
      }
    }
    ```
