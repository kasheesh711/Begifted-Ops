    ```typescript
    // Proves Pitfall 1 mitigation: Wisenet returning "2" (string) propagates as 2 (number) through pipeline.
    import { describe, expect, it } from "vitest";
    import {
      WisenetStudentsListSchema,
      WisenetSessionsListSchema,
      WisenetSessionCreditsSchema,
      MeetingStatusSchema,
    } from "@/lib/wisenet/types";
    import {
      composeDashboardSourcesFromData,
      durationMsToMinutes,
      shouldCountAsPendingDeductionWisenet,
      type WisenetMapperData,
    } from "@/lib/wisenet/mappers";

    describe("TEST-05 — z.coerce.number() at WisenetSessionsListSchema boundary", () => {
      function envelope(sessions: unknown[]) {
        return {
          status: 200,
          message: "ok",
          data: { sessions, count: sessions.length },
        };
      }
      function baseSession(overrides: Record<string, unknown>) {
        return {
          _id: "s1",
          classId: { _id: "c1", name: "Math", subject: "Math" },
          userId: { _id: "u1", name: "Teacher" },
          scheduledStartTime: "2026-04-21T09:00:00Z",
          meetingStatus: "ENDED",
          duration: 3_600_000,
          students: ["st1"],
          ...overrides,
        };
      }

      it.each([
        ["2", 2],
        [2, 2],
        ["2.5", 2.5],
        [2.5, 2.5],
        ["0", 0],
        [0, 0],
        ["360000", 360_000],
        [360_000, 360_000],
      ])("duration %s → %s (z.coerce.number normalizes string and number identically)", (raw, expected) => {
        const payload = envelope([baseSession({ duration: raw })]);
        const parsed = WisenetSessionsListSchema.parse(payload);
        expect(parsed.data.sessions[0].duration).toBe(expected);
      });
    ```

    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    # TEST-03 carve-out lint per D-23 / RESEARCH.md §TEST-03 placement.
    # Phase 2 does NOT test cache-invalidation at runtime (infra doesn't exist yet).
    # This static lint prevents new code paths from introducing the revalidateTag(..., "max") anti-pattern
    # flagged in CONCERNS.md (MEDIUM severity). Phase 3 SVC-02 rewires service.ts and removes the existing
    # occurrence; when that lands, this lint allows zero occurrences.
    #
    # Allowlist: known-safe existing occurrences in lib/dashboard/service.ts are tolerated through Phase 2.
    # The lint FAILS on occurrences in any other file.

    REPO_ROOT="$(cd "$(dirname "$0")/.." && pwd)"
    SRC_DIR="$REPO_ROOT/src"

    # Allowlist — files where existing occurrences are tolerated until Phase 3 rewrites them
    ALLOWLIST=(
      "src/lib/dashboard/service.ts"
    )

    # Grep for pattern: revalidateTag(<anything>, "max") OR revalidateTag(<anything>, 'max')
    PATTERN='revalidateTag\(.*,\s*["'"'"']max["'"'"']\)'

    echo "Scanning $SRC_DIR for revalidateTag(..., 'max') anti-pattern..."

    # Find all matches relative to web/
    cd "$REPO_ROOT"
    MATCHES=$(grep -rn -E "$PATTERN" src/ 2>/dev/null || true)

    if [ -z "$MATCHES" ]; then
      echo "✓ No revalidateTag(..., 'max') occurrences found — lint passes"
      exit 0
    fi

    # Check each match against the allowlist
    FAIL=0
    echo "$MATCHES" | while read -r line; do
      FILE_PATH=$(echo "$line" | cut -d: -f1)
      ALLOWED=0
      for a in "${ALLOWLIST[@]}"; do
        if [ "$FILE_PATH" = "$a" ]; then
          ALLOWED=1
          break
        fi
      done
      if [ "$ALLOWED" = "0" ]; then
        echo "✗ FAIL: revalidateTag(..., 'max') anti-pattern in non-allowlisted file:"
        echo "        $line"
        FAIL=1
      fi
    done

    # Re-run without subshell to determine final exit code
    NON_ALLOWLISTED=$(echo "$MATCHES" | while read -r line; do
      FILE_PATH=$(echo "$line" | cut -d: -f1)
      ALLOWED=0
      for a in "${ALLOWLIST[@]}"; do
        if [ "$FILE_PATH" = "$a" ]; then
          ALLOWED=1
          break
        fi
      done
      if [ "$ALLOWED" = "0" ]; then
        echo "$line"
      fi
    done)

    if [ -n "$NON_ALLOWLISTED" ]; then
      echo ""
      echo "CONCERNS.md flags this as a MEDIUM-severity bug. Fix: use revalidateTag(tag) (single arg) instead."
      echo "Phase 3 SVC-02 rewrites service.ts; until then new code paths must not add this pattern."
      exit 1
    fi

    echo "✓ All revalidateTag(..., 'max') occurrences are in the allowlist (tolerated through Phase 2)"
    exit 0
    ```
