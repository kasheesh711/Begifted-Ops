    #!/usr/bin/env tsx
    // DB-08 migration runner per D-22.
    // Invoked by: GH Actions CI workflow (push-to-main) or manually via `npm run db:migrate`.
    // Uses DATABASE_URL_UNPOOLED (direct Neon URL) — PgBouncer is incompatible with advisory locks and CREATE TABLE metadata (Pitfall 3).
    import { drizzle } from "drizzle-orm/neon-http";
    import { migrate } from "drizzle-orm/neon-http/migrator";
    import { neon } from "@neondatabase/serverless";

    async function main() {
      const url = process.env.DATABASE_URL_UNPOOLED;
      if (!url) {
        console.error("DATABASE_URL_UNPOOLED is required. Vercel Marketplace Neon integration auto-injects this in prod/preview; for local runs, pull via `vercel env pull web/.env`.");
        process.exit(1);
      }

      // Redact URL for log safety
      const redactedUrl = url.replace(/:[^:@]*@/, ":<REDACTED>@");
      console.log(`→ Running migrations against ${redactedUrl}`);

      const sqlClient = neon(url);
      const db = drizzle({ client: sqlClient });

      try {
        await migrate(db, { migrationsFolder: "./drizzle" });
        console.log("✓ Migrations applied successfully");
      } catch (err) {
        console.error("✗ Migration failed:", err);
        process.exit(1);
      }
    }

    main();
    ```

    ```typescript
    #!/usr/bin/env tsx
    // D-21 one-off cutover seed.
    // Runs `clasp run buildStudentAdminOwnershipMap` against the live Apps Script project,
    // validates the returned JSON against the admin-key allowlist, then bulk-inserts into
    // student_admin_ownership via Drizzle. Idempotent on student_key (ON CONFLICT DO UPDATE).
    //
    // Fallback: on clasp-run failure, writes received-or-empty JSON to
    // .planning/research/admin-ownership-seed.json for manual review and exits non-zero.

    import { execSync } from "node:child_process";
    import { writeFileSync, mkdirSync } from "node:fs";
    import { dirname, resolve } from "node:path";
    import { drizzle } from "drizzle-orm/neon-http";
    import { neon } from "@neondatabase/serverless";
    import { sql } from "drizzle-orm";
    import { studentAdminOwnership } from "../src/lib/db/schema";
    import { ADMIN_OWNER_REGISTRY, UNASSIGNED_ADMIN_KEY } from "../src/lib/dashboard/config";

    const FALLBACK_PATH = resolve("..", ".planning/research/admin-ownership-seed.json");
    const SEED_ACTOR = process.env.SEED_ACTOR_EMAIL ?? "seed@begifted-ops.local";

    const VALID_ADMIN_KEYS = new Set([...ADMIN_OWNER_REGISTRY.map((a) => a.key), UNASSIGNED_ADMIN_KEY]);

    interface OwnershipJsonEntry {
      adminKey: string;
      source?: string;
    }
    type OwnershipJson = Record<string, OwnershipJsonEntry>;

    function writeFallback(body: unknown) {
      try {
        mkdirSync(dirname(FALLBACK_PATH), { recursive: true });
      } catch { /* ignore */ }
      writeFileSync(FALLBACK_PATH, JSON.stringify(body, null, 2));
      console.error(`✗ Wrote fallback JSON to ${FALLBACK_PATH} — review before re-running`);
    }

    async function main() {
      const url = process.env.DATABASE_URL_UNPOOLED;
      if (!url) {
        console.error("DATABASE_URL_UNPOOLED is required — pull via `vercel env pull web/.env`");
        process.exit(1);
      }

      // Step 1 — clasp run
      let claspOutput: string;
      try {
        claspOutput = execSync("clasp run buildStudentAdminOwnershipMap", {
          cwd: resolve(".."), // repo root where .clasp.json + .clasprc.local.json live
          encoding: "utf-8",
          stdio: ["ignore", "pipe", "pipe"],
        });
      } catch (e) {
        console.error("✗ clasp run failed — is .clasprc.local.json present + authorized?");
        writeFallback({ error: String(e), partial: null });
        process.exit(1);
      }

      // Step 2 — parse
      let ownership: OwnershipJson;
      try {
        ownership = JSON.parse(claspOutput) as OwnershipJson;
      } catch (e) {
        console.error("✗ clasp output is not valid JSON");
        writeFallback({ error: String(e), claspOutput });
        process.exit(1);
      }

      if (!ownership || typeof ownership !== "object" || Object.keys(ownership).length === 0) {
        console.error("✗ clasp returned empty ownership map — sanity check Apps Script function");
        writeFallback({ error: "empty-or-invalid-shape", ownership });
        process.exit(1);
      }

      // Step 3 — allowlist validation
      for (const [studentKey, entry] of Object.entries(ownership)) {
        if (!VALID_ADMIN_KEYS.has(entry.adminKey)) {
          console.error(`✗ Invalid adminKey for ${studentKey}: ${entry.adminKey}`);
          console.error(`  Allowed: ${[...VALID_ADMIN_KEYS].join(", ")}`);
          writeFallback({ error: "invalid-admin-key", studentKey, entry, ownership });
          process.exit(1);
        }
      }

      console.log(`✓ Validated ${Object.keys(ownership).length} rows against allowlist`);

      // Step 4 — bulk upsert via Drizzle
      const sqlClient = neon(url);
      const db = drizzle({ client: sqlClient });

      const values = Object.entries(ownership).map(([studentKey, entry]) => ({
        studentKey,
        adminKey: entry.adminKey as "palm" | "kem" | "care" | "aya" | "petchy" | "muk" | "unassigned",
        assignedByEmail: SEED_ACTOR,
      }));

      // Chunk in batches of 500 to avoid PostgreSQL parameter-count limits on very large seeds
      const BATCH = 500;
      let inserted = 0;
      for (let i = 0; i < values.length; i += BATCH) {
        const chunk = values.slice(i, i + BATCH);
        await db
          .insert(studentAdminOwnership)
          .values(chunk)
          .onConflictDoUpdate({
            target: studentAdminOwnership.studentKey,
            set: {
              adminKey: sql`excluded.admin_key`,
              assignedByEmail: sql`excluded.assigned_by_email`,
              updatedAt: sql`now()`,
            },
          });
        inserted += chunk.length;
        console.log(`  upserted ${inserted}/${values.length} rows`);
      }

      console.log(`✓ Seeded ${inserted} admin-ownership rows`);
    }

    main().catch((err) => {
      console.error("✗ seed failed:", err);
      process.exit(1);
    });
    ```

    # DB Operations

    Two scripts live under `web/scripts/`. Both are invoked via `tsx` and read env vars.

    ## scripts/db-migrate.ts (DB-08)

    Applies pending Drizzle migrations from `web/drizzle/` against the target database.

    ### Env vars
    - `DATABASE_URL_UNPOOLED` — direct Neon URL (unpooled). The Vercel Marketplace Neon integration auto-injects this across prod/preview/dev. For local runs, pull via:
      ```bash
      vercel env pull web/.env
      ```

    ### Invocation
    - **Locally:** `cd web && npm run db:migrate`
    - **CI:** GitHub Actions workflow `.github/workflows/db-migrate.yml` runs on push-to-main when `web/drizzle/**`, `web/src/lib/db/schema.ts`, or `web/scripts/db-migrate.ts` change. Also triggerable via `workflow_dispatch`.

    ### Generating new migrations
    After editing `web/src/lib/db/schema.ts`:
    ```bash
    cd web && npm run db:generate --name=<short-name>
    ```
    Drizzle-kit writes a new `web/drizzle/NNNN_<name>.sql` file. Commit the new SQL + meta/* files in the same PR as the schema change. The CI workflow picks it up on merge.

    ### Why unpooled?
    PgBouncer (pooled `DATABASE_URL`) is incompatible with advisory locks + some CREATE TABLE metadata ops. Migrations need the direct URL. Runtime HTTP + WebSocket clients use the pooled URL — different concern.

    ## scripts/seed-admin-ownership.ts (D-21)

    One-off cutover seed. Runs at Phase 3 cutover day to populate `student_admin_ownership` from the live Apps Script `buildStudentAdminOwnershipMap` function (which aggregates the current `RemainingCredits` sheet via majority vote).

    ### Env vars
    - `DATABASE_URL_UNPOOLED` (same as db-migrate)
    - `SEED_ACTOR_EMAIL` (optional) — stamped into `assignedByEmail` column. Defaults to `seed@begifted-ops.local`.

    ### Prerequisites
    - `.clasprc.local.json` present at repo root (gitignored; your local clasp auth)
    - Global `clasp` binary installed (same baseline as existing Apps Script dev flow)
    - Apps Script project `1kAv9ICE5DQqE17JWynXJoV2Lu2bgs7pjzAfDzLNZ-1aMjhq_yKky1MQ6` authorized for Sheets access

    ### Invocation
    ```bash
    cd web && npm run db:seed-admin
    ```

    ### What happens
    1. `clasp run buildStudentAdminOwnershipMap` against the live Apps Script project
    2. Parse returned JSON (expected shape: `{ [studentKey]: { adminKey, source } }`)
    3. Validate every `adminKey` against the allowlist (palm/kem/care/aya/petchy/muk/unassigned) — fails loudly on any invalid value
    4. Drizzle `INSERT ... ON CONFLICT (student_key) DO UPDATE` — idempotent; re-running reconciles Sheet → Postgres drift

    ### Fallback
    If `clasp run` fails (auth drift, Apps Script offline, invalid response), the script writes the partial/error state to `.planning/research/admin-ownership-seed.json` for manual review and exits with code 1. Nothing is inserted.

    ### Idempotency warning
    Re-running after Phase 3's dashboard UI introduces admin-key edits will OVERWRITE dashboard edits with Sheet state. The seed is operator-run; don't automate it. Phase 5 retires the Sheet; after that, the seed should be marked deprecated.

    ## CI workflow: .github/workflows/db-migrate.yml (DB-08)

    Runs on:
    - `push` to `main` when paths match the schema/migration glob
    - `workflow_dispatch` (manual trigger via GitHub UI)

    Deploy sequencing: CI-job migration runs BEFORE Vercel deploy promotes new code. If the migration fails, the deploy doesn't land — runtime schema stays consistent.
