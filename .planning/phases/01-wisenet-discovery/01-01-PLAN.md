      assert_per_tab_row_counts() {
        local map_file="$PHASE_ROOT/.planning/research/WISENET_FIELD_MAP.md"
        [ -f "$map_file" ] || { skip_or_fail "per_tab_row_counts" "field map missing"; return; }

        # Expected counts (from web/src/lib/dashboard/config.ts REQUIRED_COLUMNS)
        local expected_aggregations=5
        local expected_credit_control=8
        local expected_upcoming=5
        local expected_students=2
        local expected_students_courses=3
        local expected_remaining_credits=2

        # awk script: for each tab heading, count data rows until the next ## heading.
        # A data row matches ^| but is not the header row (contains "Dashboard Field" or "Classification" or similar)
        # and not the separator (^\|[-:\| ]+\|\s*$).
        _count_rows_in_section() {
          local section_name="$1"
          awk -v section="$section_name" '
            $0 ~ "^## " section "$" { in_section = 1; next }
            in_section && /^## / { in_section = 0 }
            in_section && /^\|/ {
              # skip separator row (only dashes/colons/pipes/spaces)
              if ($0 ~ /^\|[-:\| ]+\|?[[:space:]]*$/) next
              # skip header row (contains "Dashboard Field" literal — which the template mandates as column 1)
              if ($0 ~ /Dashboard Field/) next
              count++
            }
            END { print count + 0 }
          ' "$map_file"
        }

        local fail_local=0
        for pair in "Aggregations:$expected_aggregations" \
                    "Credit_Control:$expected_credit_control" \
                    "Upcoming Sessions:$expected_upcoming" \
                    "Students:$expected_students" \
                    "Students & Courses:$expected_students_courses" \
                    "RemainingCredits:$expected_remaining_credits"; do
          local section="${pair%:*}"
          local expected="${pair#*:}"
          local actual
          actual="$(_count_rows_in_section "$section")"
          if [ "$actual" != "$expected" ]; then
            echo "FAIL: per_tab_row_counts — section '$section' has $actual rows, expected $expected"
            fail_local=1
          fi
        done

        if [ "$fail_local" = "0" ]; then
          echo "OK: per_tab_row_counts — all 6 tabs have exact expected row counts (5+8+5+2+3+2=25)"
        else
          FAIL_COUNT=$((FAIL_COUNT + 1))
        fi
      }
