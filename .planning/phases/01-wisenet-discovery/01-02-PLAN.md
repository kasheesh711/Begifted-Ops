<!-- Postman Collection v2.1 JSON schema shape — the parser walks this recursively.
     Source: schema.postman.com/collection/json/v2.1.0/draft-07/collection.json (cited in RESEARCH.md §Postman Collection Parsing Strategy). -->

```typescript
interface PostmanV21Collection {
  info: { name: string; schema: string; _postman_id?: string };
  item: PostmanItem[];
  auth?: PostmanAuth;
  event?: unknown[];
  variable?: Array<{ key: string; value: string; type?: string }>;
}

interface PostmanItem {
  name: string;
  item?: PostmanItem[]; // present on folders
  request?: {
    method: string;
    header: Array<{ key: string; value: string; type?: string; disabled?: boolean }>;
    url: string | {
      raw: string;
      host?: string[];
      path?: string[];
      query?: Array<{ key: string; value: string; disabled?: boolean }>;
      variable?: Array<{ key: string; value: string }>;
    };
    body?: { mode: string; raw?: string; options?: unknown };
    auth?: PostmanAuth;
  };
  response?: Array<{ name: string; status: string; body: string; header?: unknown[] }>;
}

interface PostmanAuth {
  type: "bearer" | "apikey" | "basic" | "oauth2" | "awsv4" | "digest" | "hawk" | "ntlm" | string;
  apikey?: Array<{ key: string; value: string; type?: string }>;
  bearer?: Array<{ key: string; value: string; type?: string }>;
  basic?: Array<{ key: string; value: string; type?: string }>;
  oauth2?: unknown[];
}
```

    ```typescript
    interface ParsedEndpoint {
      folder: string;      // e.g. "Students / List"
      name: string;        // e.g. "Get Students"
      method: string;      // GET, POST, etc. (always uppercased)
      path: string;        // URL with variables RESOLVED except secrets (e.g. "https://api.wisenet.co/v1/students?skip=0&take=100")
      auth: string;        // inherited or per-request auth type (e.g. "bearer+apikey", "apikey", "basic", "inherit")
      required_headers: string[];   // enabled headers, excluding Authorization + x-api-key (which show up in the auth column)
      query_params: string[];       // enabled query param keys
      body_mode: string | null;     // "raw" | "formdata" | "urlencoded" | null
      has_example_response: boolean;
    }
    ```

    **Helper functions:**

    1. `isSecretLike(value: string): boolean` — returns `true` if value length ≥ 24 AND matches `/^[A-Za-z0-9_-]+$/`. Also returns `true` if the value starts with `sk_`, `pk_`, or `Bearer ` (prefix-based). This is the check that preserves secrets as symbolic `{{name}}`.

    2. `buildVariableMap(variables: Array<{key, value}> = []): Record<string, string>` — constructs the resolution dictionary. For each variable, check `isSecretLike(value)`; if true, map `key → "__PRESERVE_SYMBOLIC__"` (special sentinel); otherwise `key → value`.

    3. `resolveString(raw: string, varMap: Record<string, string>): string` — replaces every `{{name}}` occurrence. If `varMap[name] === "__PRESERVE_SYMBOLIC__"`, leave as `{{name}}` literal. If variable missing, leave as `{{name}}` literal.

    4. `resolveUrl(url: string | PostmanUrlObject, varMap: Record<string, string>): string` — if string, call `resolveString`. If object, prefer `raw` field, call `resolveString` on it. Strip leading/trailing whitespace.

    5. `walkItems(items: PostmanItem[], folderPath: string[], inheritedAuth: PostmanAuth | undefined, varMap: Record<string, string>): ParsedEndpoint[]` — recursive. If `item.item` is array (folder), recurse with appended folder name and bubbled-down `inheritedAuth` (respecting per-folder auth overrides if present on `item.auth`). If `item.request` is object (leaf endpoint), construct `ParsedEndpoint`.

    6. `describeAuth(auth: PostmanAuth | undefined): string` — returns a human-readable string like `"bearer+apikey"`, `"apikey (x-api-key)"`, `"basic"`, `"oauth2"`, `"inherit (collection-level)"`. If apikey auth has `in: "header"`, include the header name: `"apikey (x-api-key in header)"`.

    7. `redactOutput(s: string): string` — final pass before write. Run a regex scan `/([A-Za-z0-9_-]{24,})/g` over the output. If ANY match is found AND the match is NOT already a `{{...}}` placeholder AND NOT inside a code fence, throw: `Refusing to emit WISENET_ENDPOINTS.md — found potential secret: <first 6 chars>...<last 4 chars> in folder <folder>`. (Truncate the match in the error to avoid logging the whole secret.)

    8. `emitMarkdown(endpoints: ParsedEndpoint[], collection: PostmanV21Collection, varMap: Record<string, string>): string` — builds the markdown output:

    **main() flow:**

    ```typescript
    async function main() {
      const json = readFileSync(POSTMAN_JSON_PATH, "utf8");
      const collection = JSON.parse(json) as PostmanV21Collection;

      if (!collection.info?.schema?.includes("v2.1")) {
        throw new Error(`Expected Postman Collection v2.1 schema, got: ${collection.info?.schema}`);
      }

      const varMap = buildVariableMap(collection.variable);
      const endpoints = walkItems(collection.item, [], collection.auth, varMap);

      // Sort deterministically: folder then method+path
      endpoints.sort((a, b) => {
        if (a.folder !== b.folder) return a.folder.localeCompare(b.folder);
        const akp = `${a.method} ${a.path}`;
        const bkp = `${b.method} ${b.path}`;
        return akp.localeCompare(bkp);
      });

      if (endpoints.length === 0) {
        throw new Error("No endpoints parsed from Postman collection. Check JSON shape.");
      }

      const markdown = emitMarkdown(endpoints, collection, varMap);
      const redacted = redactOutput(markdown);  // throws if secret detected

      mkdirSync(dirname(ENDPOINTS_MD_PATH), { recursive: true });
      writeFileSync(ENDPOINTS_MD_PATH, redacted, "utf8");

      console.log(`Wrote ${endpoints.length} endpoints to ${ENDPOINTS_MD_PATH}`);
      console.log(`Folders: ${Array.from(new Set(endpoints.map((e) => e.folder.split(" / ")[0]))).join(", ")}`);
    }

    void main().catch((err) => {
      console.error("Parse failed:", err);
      process.exitCode = 1;
    });
    ```
